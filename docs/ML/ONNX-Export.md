<span>### 1.1 Complete Export Script</span>
<span></span>
<span>```python</span>
<span># scripts/export_model.py</span>
<span></span>
<span>import joblib</span>
<span>import numpy as np</span>
<span>import onnx</span>
<span>import onnxruntime as ort</span>
<span>from onnxmltools.convert import convert_xgboost</span>
<span>from onnxmltools.convert.common.data_types import FloatTensorType</span>
<span>import pandas as pd</span>
<span>from pathlib import Path</span>
<span></span>
<span>class ONNXExporter:</span>
<span>    """</span>
<span>    Export XGBoost model to ONNX format for mobile deployment.</span>
<span>    """</span>
<span>    </span>
<span>    def __init__(self):</span>
<span>        self.model = None</span>
<span>        self.feature_names = None</span>
<span>        self.sample_input = None</span>
<span>    </span>
<span>    def load_model(self, model_path: str):</span>
<span>        """</span>
<span>        Load trained XGBoost model.</span>
<span>        </span>
<span>        Args:</span>
<span>            model_path: Path to .pkl model file</span>
<span>        """</span>
<span>        </span>
<span>        data = joblib.load(model_path)</span>
<span>        </span>
<span>        # Handle both direct model and wrapped model</span>
<span>        if isinstance(data, dict):</span>
<span>            self.model = data['model']</span>
<span>            self.feature_names = data.get('feature_names')</span>
<span>        else:</span>
<span>            self.model = data</span>
<span>        </span>
<span>        print(f"✅ Loaded model from {model_path}")</span>
<span>        print(f"   Model type: {type(self.model).__name__}")</span>
<span>        </span>
<span>        return self</span>
<span>    </span>
<span>    def create_sample_input(self, n_samples: int = 1) -&gt; np.ndarray:</span>
<span>        """</span>
<span>        Create sample input for shape inference.</span>
<span>        </span>
<span>        Args:</span>
<span>            n_samples: Number of sample rows</span>
<span>        </span>
<span>        Returns:</span>
<span>            Sample input array</span>
<span>        """</span>
<span>        </span>
<span>        # 13 features (based on Feature Engineering)</span>
<span>        FEATURE_COUNT = 13</span>
<span>        </span>
<span>        # Create realistic sample values</span>
<span>        sample = np.array([[</span>
<span>            2.5,    # crime_density</span>
<span>            2.0,    # crime_severity</span>
<span>            0.5,    # hour_sin</span>
<span>            0.866,  # hour_cos</span>
<span>            0.0,    # day_of_week_sin</span>
<span>            1.0,    # day_of_week_cos</span>
<span>            0.5,    # month_sin</span>
<span>            0.866,  # month_cos</span>
<span>            0,      # is_weekend</span>
<span>            1,      # is_nighttime</span>
<span>            0,      # is_rush_hour</span>
<span>            0.6,    # lighting_risk</span>
<span>            0.3,    # crowd_risk</span>
<span>            0.1     # weather_risk</span>
<span>        ]] * n_samples, dtype=np.float32)</span>
<span>        </span>
<span>        self.sample_input = sample</span>
<span>        return sample</span>
<span>    </span>
<span>    def export_to_onnx(</span>
<span>        self,</span>
<span>        output_path: str,</span>
<span>        optimize: bool = True,</span>
<span>        quantize: bool = False</span>
<span>    ):</span>
<span>        """</span>
<span>        Export model to ONNX format.</span>
<span>        </span>
<span>        Args:</span>
<span>            output_path: Where to save .onnx file</span>
<span>            optimize: Apply ONNX optimizations</span>
<span>            quantize: Quantize to int8 for mobile</span>
<span>        """</span>
<span>        </span>
<span>        if self.model is None:</span>
<span>            raise ValueError("No model loaded. Call load_model() first.")</span>
<span>        </span>
<span>        if self.sample_input is None:</span>
<span>            self.create_sample_input()</span>
<span>        </span>
<span>        print("Exporting model to ONNX...")</span>
<span>        </span>
<span>        # Define input shape</span>
<span>        initial_types = [</span>
<span>            ('input', FloatTensorType([None, self.sample_input.shape[1]]))</span>
<span>        ]</span>
<span>        </span>
<span>        # Convert to ONNX</span>
<span>        onnx_model = convert_xgboost(</span>
<span>            self.model,</span>
<span>            initial_types=initial_types,</span>
<span>            target_opset=12  # Broad compatibility</span>
<span>        )</span>
<span>        </span>
<span>        # Optimize if requested</span>
<span>        if optimize:</span>
<span>            from onnx import optimizer</span>
<span>            </span>
<span>            passes = [</span>
<span>                'fuse_consecutive_transposes',</span>
<span>                'eliminate_deadend',</span>
<span>                'eliminate_identity',</span>
<span>                'eliminate_nop_transpose',</span>
<span>                'eliminate_nop_pad',</span>
<span>                'fuse_transpose_into_gemm'</span>
<span>            ]</span>
<span>            </span>
<span>            onnx_model = optimizer.optimize(onnx_model, passes=passes)</span>
<span>            print("   ✓ Optimized")</span>
<span>        </span>
<span>        # Save initial model</span>
<span>        onnx.save(onnx_model, output_path)</span>
<span>        print(f"   ✓ Saved to {output_path}")</span>
<span>        </span>
<span>        # Quantize if requested (smaller file, faster inference)</span>
<span>        if quantize:</span>
<span>            quantized_path = output_path.replace('.onnx', '_quantized.onnx')</span>
<span>            self._quantize_model(output_path, quantized_path)</span>
<span>            print(f"   ✓ Quantized model saved to {quantized_path}")</span>
<span>        </span>
<span>        # Validate</span>
<span>        self._validate_export(output_path)</span>
<span>        </span>
<span>        return onnx_model</span>
<span>    </span>
<span>    def _quantize_model(self, input_path: str, output_path: str):</span>
<span>        """</span>
<span>        Quantize ONNX model to int8.</span>
<span>        </span>
<span>        Args:</span>
<span>            input_path: Path to FP32 ONNX model</span>
<span>            output_path: Path for quantized model</span>
<span>        """</span>
<span>        </span>
<span>        from onnxruntime.quantization import quantize_dynamic, QuantType</span>
<span>        </span>
<span>        quantize_dynamic(</span>
<span>            model_input=input_path,</span>
<span>            model_output=output_path,</span>
<span>            weight_type=QuantType.QInt8,</span>
<span>            optimize_model=True</span>
<span>        )</span>
<span>    </span>
<span>    def _validate_export(self, onnx_path: str):</span>
<span>        """</span>
<span>        Validate exported model produces same results.</span>
<span>        </span>
<span>        Args:</span>
<span>            onnx_path: Path to exported ONNX model</span>
<span>        """</span>
<span>        </span>
<span>        # Load ONNX model</span>
<span>        ort_session = ort.InferenceSession(onnx_path)</span>
<span>        </span>
<span>        # Run inference with original model</span>
<span>        original_preds = self.model.predict(self.sample_input)</span>
<span>        </span>
<span>        # Run inference with ONNX</span>
<span>        onnx_input = {'input': self.sample_input}</span>
<span>        onnx_output = ort_session.run(None, onnx_input)</span>
<span>        onnx_preds = onnx_output[0]</span>
<span>        </span>
<span>        # Compare results</span>
<span>        if hasattr(original_preds, 'flatten'):</span>
<span>            original_preds = original_preds.flatten()</span>
<span>        </span>
<span>        if hasattr(onnx_preds, 'flatten'):</span>
<span>            onnx_preds = onnx_preds.flatten()</span>
<span>        </span>
<span>        # Check if predictions match</span>
<span>        match = np.array_equal(original_preds, onnx_preds)</span>
<span>        </span>
<span>        if match:</span>
<span>            print("   ✓ Validation passed: Original and ONNX predictions match")</span>
<span>        else:</span>
<span>            print("   ⚠️ Validation warning: Predictions differ")</span>
<span>            print(f"      Original: {original_preds[:5]}")</span>
<span>            print(f"      ONNX: {onnx_preds[:5]}")</span>
<span>    </span>
<span>    def get_model_info(self, onnx_path: str) -&gt; dict:</span>
<span>        """</span>
<span>        Get information about exported model.</span>
<span>        </span>
<span>        Returns:</span>
<span>            Dictionary with model info</span>
<span>        """</span>
<span>        </span>
<span>        model = onnx.load(onnx_path)</span>
<span>        </span>
<span>        # Get input shape</span>
<span>        input_shape = None</span>
<span>        for inp in model.graph.input:</span>
<span>            input_shape = [dim.dim_value for dim in inp.type.tensor_type.shape.dim]</span>
<span>            break</span>
<span>        </span>
<span>        return {</span>
<span>            'input_shape': input_shape,</span>
<span>            'opset_version': model.opset_import[0].version,</span>
<span>            'graph_nodes': len(model.graph.node),</span>
<span>            'file_size_mb': Path(onnx_path).stat().st_size / (1024 * 1024)</span>
<span>        }</span>
<span></span>
<span></span>
<span># Main execution</span>
<span>def main():</span>
<span>    """Command-line entry point."""</span>
<span>    </span>
<span>    import argparse</span>
<span>    </span>
<span>    parser = argparse.ArgumentParser(description='Export XGBoost model to ONNX')</span>
<span>    parser.add_argument('--input', type=str, default='models/production/risk_model.pkl',</span>
<span>                        help='Input model path (.pkl)')</span>
<span>    parser.add_argument('--output', type=str, default='models/production/risk_model.onnx',</span>
<span>                        help='Output path (.onnx)')</span>
<span>    parser.add_argument('--quantize', action='store_true',</span>
<span>                        help='Create quantized version for mobile')</span>
<span>    parser.add_argument('--no-optimize', action='store_true',</span>
<span>                        help='Skip ONNX optimizations')</span>
<span>    </span>
<span>    args = parser.parse_args()</span>
<span>    </span>
<span>    print("=" * 60)</span>
<span>    print("ONNX EXPORTER")</span>
<span>    print("=" * 60)</span>
<span>    </span>
<span>    exporter = ONNXExporter()</span>
<span>    exporter.load_model(args.input)</span>
<span>    exporter.create_sample_input()</span>
<span>    </span>
<span>    exporter.export_to_onnx(</span>
<span>        output_path=args.output,</span>
<span>        optimize=not args.no_optimize,</span>
<span>        quantize=args.quantize</span>
<span>    )</span>
<span>    </span>
<span>    # Print model info</span>
<span>    info = exporter.get_model_info(args.output)</span>
<span>    print("\nModel Info:")</span>
<span>    print(f"  Input shape: {info['input_shape']}")</span>
<span>    print(f"  Opset version: {info['opset_version']}")</span>
<span>    print(f"  Graph nodes: {info['graph_nodes']}")</span>
<span>    print(f"  File size: {info['file_size_mb']:.2f} MB")</span>
<span>    </span>
<span>    if args.quantize:</span>
<span>        quantized_path = args.output.replace('.onnx', '_quantized.onnx')</span>
<span>        info_q = exporter.get_model_info(quantized_path)</span>
<span>        print(f"\nQuantized Model Info:")</span>
<span>        print(f"  File size: {info_q['file_size_mb']:.2f} MB")</span>
<span>        print(f"  Compression: {(1 - info_q['file_size_mb']/info['file_size_mb'])*100:.1f}% smaller")</span>
<span></span>
